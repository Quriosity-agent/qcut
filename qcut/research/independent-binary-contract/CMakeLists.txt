cmake_minimum_required(VERSION 3.16)
project(independent_binary_contracts LANGUAGES CXX)

option(BINARY_CONTRACT_SANITIZERS "Enable fail-closed ASan and UBSan for all contracts" OFF)
option(BINARY_CONTRACT_NATIVE_PROBES "Compile optional macOS arm64 diagnostics" OFF)

enable_testing()
foreach(contract AGFX EDITOR CREATOR LENS FOG)
    set(${contract}_CONTRACT_SANITIZERS ${BINARY_CONTRACT_SANITIZERS} CACHE BOOL "" FORCE)
endforeach()
foreach(contract AGFX EDITOR CREATOR)
    set(${contract}_CONTRACT_NATIVE_PROBE ${BINARY_CONTRACT_NATIVE_PROBES} CACHE BOOL "" FORCE)
endforeach()
set(LENS_CONTRACT_NATIVE_ORACLE ${BINARY_CONTRACT_NATIVE_PROBES} CACHE BOOL "" FORCE)
set(SOFT_GLOW_SANITIZERS ${BINARY_CONTRACT_SANITIZERS} CACHE BOOL "" FORCE)
set(SOFT_GLOW_NATIVE_PROBE ${BINARY_CONTRACT_NATIVE_PROBES} CACHE BOOL "" FORCE)

foreach(contract agfx editor creator lens)
    add_subdirectory(../independent-${contract}-contract ${contract})
endforeach()
add_subdirectory(../independent-soft-glow soft-glow)
add_subdirectory(../independent-fog-contract fog)

if(BINARY_CONTRACT_NATIVE_PROBES)
    add_library(cgl-pass-capture SHARED ../jianying-runtime-probe/cgl-pass-capture.cpp)
    target_compile_features(cgl-pass-capture PRIVATE cxx_std_20)
    target_compile_options(cgl-pass-capture PRIVATE -Wall -Wextra -Wpedantic -Werror -Wno-deprecated-declarations)
    target_link_libraries(cgl-pass-capture PRIVATE "-framework OpenGL")
endif()
