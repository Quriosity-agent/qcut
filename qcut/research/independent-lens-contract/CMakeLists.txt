cmake_minimum_required(VERSION 3.16)
project(independent_lens_contract LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

option(LENS_CONTRACT_SANITIZERS "Enable address/undefined sanitizers" OFF)
option(LENS_CONTRACT_NATIVE_ORACLE "Build the optional macOS arm64 diagnostic" OFF)

if(LENS_CONTRACT_SANITIZERS AND MSVC)
  message(FATAL_ERROR "Address/undefined sanitizer combination is not configured for MSVC")
endif()

add_library(lens_contract STATIC matrix.cpp points.cpp gaussian.cpp image_warp.cpp transform_plan.cpp temporal_crop.cpp crop_selection.cpp)
target_compile_features(lens_contract PUBLIC cxx_std_20)
target_include_directories(lens_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(lens-contract-tests contract_tests.cpp)
target_link_libraries(lens-contract-tests PRIVATE lens_contract)
add_executable(lens-contract main.cpp)
target_link_libraries(lens-contract PRIVATE lens_contract)
add_executable(lens-image-warp-tests image_warp_tests.cpp)
target_link_libraries(lens-image-warp-tests PRIVATE lens_contract)
add_executable(lens-image-warp image_warp_main.cpp)
target_link_libraries(lens-image-warp PRIVATE lens_contract)
add_executable(lens-image-warp-backend-tests image_warp_backend_tests.cpp)
target_link_libraries(lens-image-warp-backend-tests PRIVATE lens_contract)
add_executable(lens-transform-plan-tests transform_plan_tests.cpp)
target_link_libraries(lens-transform-plan-tests PRIVATE lens_contract)
add_executable(lens-temporal-crop-tests temporal_crop_tests.cpp)
target_link_libraries(lens-temporal-crop-tests PRIVATE lens_contract)
add_executable(lens-temporal-crop temporal_crop_main.cpp)
target_link_libraries(lens-temporal-crop PRIVATE lens_contract)
set(contract_targets lens_contract lens-contract-tests lens-contract lens-image-warp-tests lens-image-warp lens-image-warp-backend-tests lens-transform-plan-tests lens-temporal-crop-tests lens-temporal-crop)
add_executable(lens-crop-selection-tests crop_selection_tests.cpp)
target_link_libraries(lens-crop-selection-tests PRIVATE lens_contract)
add_executable(lens-crop-selection crop_selection_main.cpp)
target_link_libraries(lens-crop-selection PRIVATE lens_contract)
list(APPEND contract_targets lens-crop-selection-tests lens-crop-selection)

if(LENS_CONTRACT_NATIVE_ORACLE)
  if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
    message(FATAL_ERROR "Native ABI is verified only on macOS arm64")
  endif()
  add_executable(lens-native-oracle native_oracle.cpp)
  target_link_libraries(lens-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-native-oracle)
  add_executable(lens-image-native-oracle image_warp_native_oracle.cpp)
  target_link_libraries(lens-image-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-image-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-image-native-oracle)
  add_executable(lens-warp-backend-native-oracle image_warp_backend_native_oracle.cpp)
  target_link_libraries(lens-warp-backend-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-warp-backend-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-warp-backend-native-oracle)
  add_executable(lens-transform-plan-native-oracle transform_plan_native_oracle.cpp)
  target_link_libraries(lens-transform-plan-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-transform-plan-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-transform-plan-native-oracle)
  add_executable(lens-temporal-native-oracle temporal_crop_native_oracle.cpp)
  target_link_libraries(lens-temporal-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-temporal-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-temporal-native-oracle)
  add_executable(lens-crop-selection-native-oracle crop_selection_native_oracle.cpp)
  target_link_libraries(lens-crop-selection-native-oracle PRIVATE lens_contract)
  target_compile_options(lens-crop-selection-native-oracle PRIVATE -Wno-deprecated-declarations)
  list(APPEND contract_targets lens-crop-selection-native-oracle)

foreach(target IN LISTS contract_targets)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off -fno-fast-math)
    if(LENS_CONTRACT_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
    endif()
  endif()
endforeach()
