cmake_minimum_required(VERSION 3.16)
project(independent_creator_contract LANGUAGES CXX)

option(CREATOR_CONTRACT_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(CREATOR_CONTRACT_NATIVE_PROBE "Build optional macOS arm64 native diagnostics" OFF)

add_library(creator_contract STATIC selection.cpp requests.cpp model_gate.cpp dialog_callbacks.cpp common_keyframes.cpp editor_events.cpp keyframe_insertion.cpp keyframe_controls.cpp dirty_tree.cpp)
target_compile_features(creator_contract PUBLIC cxx_std_20)
target_include_directories(creator_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")

set(contract_targets creator_contract)
if(TARGET editor_contract)
    target_link_libraries(creator_contract PUBLIC editor_contract)
else()
    add_library(creator_editor_value STATIC ../independent-editor-contract/value_state.cpp ../independent-editor-contract/window.cpp)
    target_compile_features(creator_editor_value PUBLIC cxx_std_20)
    target_link_libraries(creator_contract PUBLIC creator_editor_value)
    list(APPEND contract_targets creator_editor_value)
endif()
enable_testing()
foreach(unit selection requests model_gate dialog_callbacks common_keyframes editor_events keyframe_insertion keyframe_controls dirty_tree)
    add_executable(creator-${unit}-tests ${unit}_tests.cpp)
    target_link_libraries(creator-${unit}-tests PRIVATE creator_contract)
    add_test(NAME creator-${unit} COMMAND creator-${unit}-tests)
    list(APPEND contract_targets creator-${unit}-tests)
endforeach()

if(CREATOR_CONTRACT_NATIVE_PROBE)
    if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
        message(FATAL_ERROR "The diagnostic ABI is verified only on macOS arm64")
    endif()
    enable_language(OBJCXX)
    add_executable(creator-native-dirty native_dirty.mm)
    target_link_libraries(creator-native-dirty PRIVATE creator_contract "-framework Foundation")
    target_compile_options(creator-native-dirty PRIVATE -fobjc-arc -Wno-deprecated-declarations)
    list(APPEND contract_targets creator-native-dirty)
    add_executable(creator-native-insertion native_insertion.mm)
    target_link_libraries(creator-native-insertion PRIVATE creator_contract "-framework Foundation")
    target_compile_options(creator-native-insertion PRIVATE -fobjc-arc -Wno-deprecated-declarations)
    list(APPEND contract_targets creator-native-insertion)
    add_executable(creator-native-events native_events.mm)
    target_link_libraries(creator-native-events PRIVATE creator_contract "-framework Foundation")
    target_compile_options(creator-native-events PRIVATE -fobjc-arc -Wno-deprecated-declarations)
    list(APPEND contract_targets creator-native-events)
    add_executable(creator-native-constants native_constants.cpp)
    target_link_libraries(creator-native-constants PRIVATE creator_contract)
    target_compile_options(creator-native-constants PRIVATE -Wno-deprecated-declarations)
    list(APPEND contract_targets creator-native-constants)
endif()

foreach(target IN LISTS contract_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
        if(CREATOR_CONTRACT_SANITIZERS)
            message(FATAL_ERROR "Use a Clang/GCC toolchain for combined ASan and UBSan")
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -Wconversion -Wsign-conversion -ffp-contract=off -fno-fast-math)
        if(CREATOR_CONTRACT_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        endif()
    endif()
endforeach()
