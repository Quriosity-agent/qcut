cmake_minimum_required(VERSION 3.16)
project(independent_editor_contract LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)

option(EDITOR_CONTRACT_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(EDITOR_CONTRACT_NATIVE_PROBE "Build the optional macOS arm64 diagnostic" OFF)

add_library(editor_contract STATIC value_state.cpp filter_time.cpp keyframe.cpp resample.cpp bezier.cpp)
target_compile_features(editor_contract PUBLIC cxx_std_20)
target_include_directories(editor_contract PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set(contract_targets editor_contract)

foreach(unit value_state filter_time keyframe resample bezier)
    add_executable(editor-${unit}-tests ${unit}_tests.cpp)
    target_link_libraries(editor-${unit}-tests PRIVATE editor_contract)
    list(APPEND contract_targets editor-${unit}-tests)
endforeach()

if(EDITOR_CONTRACT_NATIVE_PROBE)
    if(NOT APPLE OR NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "arm64")
        message(FATAL_ERROR "The diagnostic private ABI is only verified on macOS arm64")
    endif()
    enable_language(OBJCXX)
    foreach(probe native evaluation)
        set(target editor-${probe}-probe)
        add_executable(${target} ${probe}_probe.mm)
        set_target_properties(${target} PROPERTIES OBJCXX_STANDARD 20 OBJCXX_STANDARD_REQUIRED YES)
        target_compile_options(${target} PRIVATE -fobjc-arc -Wno-deprecated-declarations)
        target_link_libraries(${target} PRIVATE editor_contract "-framework Foundation")
        list(APPEND contract_targets ${target})
    endforeach()
endif()

foreach(target IN LISTS contract_targets)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /WX /fp:strict)
        if(EDITOR_CONTRACT_SANITIZERS)
            message(FATAL_ERROR "This sanitizer configuration requires Clang or GCC")
        endif()
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror -ffp-contract=off)
        if(EDITOR_CONTRACT_SANITIZERS)
            target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
            target_link_options(${target} PRIVATE -fsanitize=address,undefined -fno-sanitize-recover=all)
        endif()
    endif()
endforeach()

enable_testing()
foreach(unit value_state filter_time keyframe resample bezier)
    add_test(NAME editor-${unit}-contract COMMAND editor-${unit}-tests)
endforeach()
